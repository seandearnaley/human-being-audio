cmake_minimum_required(VERSION 3.25)
project(StudioGainStudy LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJC OBJCXX)
endif()
set(STUDIO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Studio checkout supplying the two engines")
set(CMAKE_CXX_STANDARD 20)
set(LIVECUT_BUILD_TESTS OFF CACHE BOOL "")
set(LIVECUT_BUILD_PLUGIN OFF CACHE BOOL "")
set(TF_BUILD_TESTS OFF CACHE BOOL "")
set(TF_BUILD_PLUGIN OFF CACHE BOOL "")
add_subdirectory("${STUDIO_SOURCE}/products/livecut-redux" livecut EXCLUDE_FROM_ALL)
add_subdirectory("${STUDIO_SOURCE}/products/tract-form" tract EXCLUDE_FROM_ALL)
add_executable(render-gain render.cpp)
target_include_directories(render-gain PRIVATE "${STUDIO_SOURCE}/products/livecut-redux/tests/engine")
target_link_libraries(render-gain PRIVATE LivecutEngine tf_engine)
