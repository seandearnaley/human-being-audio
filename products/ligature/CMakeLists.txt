cmake_minimum_required(VERSION 3.25)

set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")

project(Ligature VERSION 2.0.0 LANGUAGES C CXX)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(APPLE)
    enable_language(OBJC OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CMakeDependentOption)
cmake_dependent_option(LG_BUILD_PLUGIN "Build the interface and the plugin (fetches JUCE and Visage)" ON "NOT HBA_RTSAN" OFF)
option(LG_BUILD_TESTS "Build the tests" ON)
option(LG_INSTALL_PLUGIN "Copy the plugins into ~/Library/Audio/Plug-Ins after building" OFF)

add_library(lg_warnings INTERFACE)
target_compile_options(lg_warnings INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)

include("${PROJECT_SOURCE_DIR}/../../cmake/CPM.cmake")
include("${PROJECT_SOURCE_DIR}/../../cmake/StudioHelpers.cmake")
include("${PROJECT_SOURCE_DIR}/../../cmake/StudioDependencies.cmake")
hba_enable_ccache()
hba_add_nlohmann_json()

add_subdirectory("${PROJECT_SOURCE_DIR}/../../modules/dsp" "${PROJECT_BINARY_DIR}/studio-dsp")
add_subdirectory("${PROJECT_SOURCE_DIR}/../../modules/core" "${PROJECT_BINARY_DIR}/studio-core")
add_subdirectory(core)
if(EXISTS "${PROJECT_SOURCE_DIR}/engine/CMakeLists.txt")
    add_subdirectory(engine)
endif()
option(LG_BUILD_VOICES "Build the studio voices (Torus Glide, Lineage, Datum and Strike Body engines)" ON)
if(LG_BUILD_VOICES AND EXISTS "${PROJECT_SOURCE_DIR}/voices/CMakeLists.txt")
    add_subdirectory(voices)
endif()
if(EXISTS "${PROJECT_SOURCE_DIR}/tools/lg_render.cpp")
    add_subdirectory(tools)
endif()

if(LG_BUILD_PLUGIN)
    hba_add_visage()
    hba_add_juce()
    add_subdirectory("${PROJECT_SOURCE_DIR}/../../modules/visage-ui" "${PROJECT_BINARY_DIR}/studio-visage-ui")
    add_subdirectory("${PROJECT_SOURCE_DIR}/../../modules/plugin-host" "${PROJECT_BINARY_DIR}/studio-plugin-host")
    if(EXISTS "${PROJECT_SOURCE_DIR}/ui/CMakeLists.txt")
        add_subdirectory(ui)
    endif()
    if(EXISTS "${PROJECT_SOURCE_DIR}/plugin/src/PluginProcessor.cpp")
        add_subdirectory(plugin)
    endif()
endif()

if(LG_BUILD_TESTS)
    enable_testing()
    hba_add_catch2()
    list(APPEND CMAKE_MODULE_PATH "${Catch2_SOURCE_DIR}/extras")
    include(Catch)
    add_subdirectory(tests)
endif()
