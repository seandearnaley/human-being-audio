# Studio voices: Ligature's parts played through the studio's own instruments (Torus Glide, Lineage, Strike Body and
# Datum). Each instrument's core and engine is built from its own product directory; its factory presets are embedded
# at configure time so lg_voices needs no files at run time.
set(LG_STUDIO_PRODUCTS "${PROJECT_SOURCE_DIR}/..")

# The warning targets the instruments' CMakeLists expect, as each product defines them.
foreach(prefix IN ITEMS tg ln dt sb)
    if(NOT TARGET ${prefix}_warnings)
        add_library(${prefix}_warnings INTERFACE)
        target_compile_options(${prefix}_warnings INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endforeach()

foreach(product IN ITEMS torus-glide lineage datum strike-body)
    foreach(part IN ITEMS core engine)
        add_subdirectory("${LG_STUDIO_PRODUCTS}/${product}/${part}" "${PROJECT_BINARY_DIR}/studio-${product}-${part}")
    endforeach()
endforeach()

# One factory preset per voice: name in the generated header, then the product's preset file.
set(LG_VOICE_PRESETS
    chords "torus-glide/presets/28 Live Capture Blank Slate.json"
    melody "torus-glide/presets/15 MPE Slide Manual.json"
    bass "lineage/presets/01-velvet-analog-reese.json"
    arp "strike-body/presets/17 Glass Bell Perc.json"
    clap "strike-body/presets/12 Layered Neuro Clap.json"
    kick "datum/presets/10-twin-t-909-club.json")

set(lg_presets_header "${CMAKE_CURRENT_BINARY_DIR}/generated/lg/voice_presets.h")
set(lg_presets_text "// Generated by products/ligature/voices/CMakeLists.txt from the studio's factory presets. Do not edit.\n")
string(APPEND lg_presets_text "#pragma once\n\n#include <string_view>\n\nnamespace lg::voice_presets {\n")
list(LENGTH LG_VOICE_PRESETS lg_preset_items)
math(EXPR lg_preset_last "${lg_preset_items} - 1")
foreach(index RANGE 0 ${lg_preset_last} 2)
    math(EXPR path_index "${index} + 1")
    list(GET LG_VOICE_PRESETS ${index} name)
    list(GET LG_VOICE_PRESETS ${path_index} relative)
    set(path "${LG_STUDIO_PRODUCTS}/${relative}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Studio voice preset not found: ${path}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
    string(APPEND lg_presets_text "\n// ${relative}\ninline constexpr unsigned char k_${name}[] = {${bytes}};\n")
    string(APPEND lg_presets_text "inline std::string_view ${name}() noexcept {\n")
    string(APPEND lg_presets_text "    return {reinterpret_cast<const char*>(k_${name}), sizeof(k_${name})};\n}\n")
endforeach()
string(APPEND lg_presets_text "\n} // namespace lg::voice_presets\n")
file(WRITE "${lg_presets_header}.tmp" "${lg_presets_text}")
configure_file("${lg_presets_header}.tmp" "${lg_presets_header}" COPYONLY) # rewrites only when a preset changed

add_library(lg_voices STATIC
    include/lg/studio_voices.h
    src/studio_voices.cpp
    "${lg_presets_header}")

target_include_directories(lg_voices PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(lg_voices PUBLIC cxx_std_20)
target_link_libraries(lg_voices
    PUBLIC lg_engine
    PRIVATE tg_core tg_engine ln_core ln_engine dt_core dt_engine sb_core sb_engine hba_dsp lg_warnings)
hba_require_nonblocking(lg_voices)
